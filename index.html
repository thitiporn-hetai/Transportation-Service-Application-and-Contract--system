function savePdfToDrive(pdfBase64, fullName) { try { const base64Data = pdfBase64.includes(',') ? pdfBase64.split(',')[1] : pdfBase64; const blob = Utilities.newBlob( Utilities.base64Decode(base64Data), 'application/pdf', `ใบสมัครและสัญญาจ้างขนส่ง_${fullName || 'ผู้สมัคร'}.pdf` ); const folderName = 'สัญญาจ้างขนส่ง_Hetai'; const folders = DriveApp.getFoldersByName(folderName); const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return file.getUrl(); } catch (err) { return 'อัปโหลด PDF ไม่สำเร็จ: ' + err.message; } } function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const sheet = getOrCreateDataSheet(); const pdfUrl = payload.pdfBase64 ? savePdfToDrive(payload.pdfBase64, payload.fullName) : ''; const timestamp = Utilities.formatDate(new Date(), 'Asia/Bangkok', 'yyyy-MM-dd HH:mm:ss'); sheet.appendRow([ timestamp, payload.fullName || '', payload.age || '', payload.idCard || '', payload.phone || '', payload.address || '', payload.emerName || '', payload.emerRelation || '', payload.emerPhone || '', payload.position || '', payload.rate || '', payload.plateNumber || '', payload.hubLocation || '', payload.bankName || '', payload.bankAccountName || '', payload.bankAccount || '', pdfUrl, payload.employeeCode || '', payload.registeredAddress || '' ]); return ContentService .createTextOutput(JSON.stringify({ status: 'success', pdfUrl: pdfUrl })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { const sheet = getOrCreateDataSheet(); const lastRow = sheet.getLastRow(); if (lastRow < 2) { return ContentService .createTextOutput(JSON.stringify({ status: 'success', data: [] })) .setMimeType(ContentService.MimeType.JSON); } const rows = sheet.getRange(2, 1, lastRow - 1, 19).getValues(); const data = rows.map(r => ({ timestamp: r[0], fullName: r[1], age: r[2], idCard: r[3], phone: r[4], address: r[5], emerName: r[6], emerRelation: r[7], emerPhone: r[8], position: r[9], rate: r[10], plateNumber: r[11], hubLocation: r[12], bankName: r[13], bankAccountName: r[14], bankAccount: r[15], pdfUrl: r[16], employeeCode: r[17] || '', registeredAddress: r[18] || '' })).reverse(); return ContentService .createTextOutput(JSON.stringify({ status: 'success', data: data })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } function getOrCreateDataSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheetName = 'Data'; let sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); sheet.appendRow([ 'วันที่สมัคร', 'ชื่อ-นามสกุล', 'อายุ', 'เลขบัตรประชาชน', 'เบอร์โทร', 'ที่อยู่', 'บุคคลติดต่อฉุกเฉิน', 'ความสัมพันธ์', 'เบอร์ฉุกเฉิน', 'ตำแหน่ง/รถ', 'เรทค่าตอบแทน', 'ทะเบียนรถ', 'Hub', 'ธนาคาร', 'ชื่อบัญชี', 'เลขที่บัญชี', 'ลิงค์ไฟล์PDF', 'รหัสพนักงาน', 'ที่อยู่ตามทะเบียนบ้าน' ]); } return sheet; }
